THEME_PROPERTY = "gdtf_comparer_theme"
def theme_for_scheme(scheme: Qt.ColorScheme) -> str
def apply_theme(app: QApplication, theme: str) -> bool         # False when already applied
class OsThemeFollower(QObject):                                 # parented to the QApplication
